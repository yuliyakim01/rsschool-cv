                    public class nthPowerAlgs {
                        public static double bruteForce(int x, int n) {
                            double result = 1;
                            for(int i=0; i < n; i++) {
                                result *= x; 
                            }
                            return result;		
                        }
                        public static double decreaseAndConquer(int x, int n) {
                            if (n == 0) {
                                return 1;
                            }
                            double ans = decreaseAndConquer(x, n/2);
                            if (n % 2 == 0) {
                                return ans * ans;
                            }
                            else {
                                return x * (ans * ans);
                            }
                        }
                    }
